add_library(laso_options INTERFACE)
set(LASO_SANITIZER_FLAGS "")
target_compile_features(laso_options INTERFACE cxx_std_20)
target_compile_options(laso_options INTERFACE -Wall -Wextra -Wpedantic)
foreach(sanitizer IN ITEMS ASAN UBSAN TSAN)
  if(LASO_ENABLE_${sanitizer})
    if(sanitizer STREQUAL "ASAN")
      set(flag address)
    elseif(sanitizer STREQUAL "UBSAN")
      set(flag undefined)
    else()
      set(flag thread)
    endif()
    target_compile_options(laso_options INTERFACE -fsanitize=${flag} -fno-omit-frame-pointer)
    target_link_options(laso_options INTERFACE -fsanitize=${flag})
    list(APPEND LASO_SANITIZER_FLAGS -fsanitize=${flag} -fno-omit-frame-pointer)
  endif()
endforeach()

function(laso_target target)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_link_libraries(${target} PRIVATE laso_options)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>)
  if(LASO_ENABLE_TIDY)
    set_target_properties(${target} PROPERTIES CXX_CLANG_TIDY "${LASO_CLANG_TIDY}")
  endif()
endfunction()
